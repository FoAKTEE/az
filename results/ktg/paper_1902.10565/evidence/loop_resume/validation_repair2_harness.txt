#!/bin/bash
# Supplement to harness2.sh: K4 redone with a stand-in loop that has the real
# loop script's error semantics (bash -eu, pipefail), so a killed child ends the
# loop with 143 as synchronous_loop_9x9.sh would. Three orderings, wrapper never
# signalled. Same shims, same setup as harness2.sh.
set -u
AZ=/home/schmidt/ssci-haiyangw/az; SIM="${VAL_SIM:?}"; DB="$AZ/results/ktg/paper_1902.10565/codes/data_budget"; LOOPDIR="$AZ/results/ktg/paper_1902.10565/codes/loop"; REAL_SBATCH="$LOOPDIR/loop.sbatch"
rm -rf "$SIM"; mkdir -p "$SIM/bin" "$SIM/ktg/loop" "$SIM/ktg/logs" "$SIM/codes/loop" "$SIM/src"
cat > "$SIM/bin/sbatch" <<'EOF'
#!/bin/bash
n=$(cat "$VAL_JOBCTR" 2>/dev/null || echo 700000); n=$((n+1)); echo "$n" > "$VAL_JOBCTR"; echo "SBATCH[$n] $*" >> "$VAL_SHIMLOG"; echo "$n"
EOF
cat > "$SIM/bin/scancel" <<'EOF'
#!/bin/bash
echo "SCANCEL $*" >> "$VAL_SHIMLOG"
EOF
for c in sinfo sacct scontrol squeue nvidia-smi; do printf '#!/bin/bash\nexit 0\n' > "$SIM/bin/$c"; done; chmod +x "$SIM"/bin/*
export VAL_SHIMLOG="$SIM/shim.log" VAL_JOBCTR="$SIM/jobctr"; export PATH="$SIM/bin:$PATH"
ln -s "$DB" "$SIM/codes/data_budget"; printf 'export KATAGO_SRC=%s\nexport KATAGO_BIN=%s/katago\n' "$SIM/src" "$SIM/src" > "$SIM/ktg/env.sh"
export AZ_ROOT="$AZ" KTG_ROOT="$SIM/ktg" KTG_CODES="$SIM/codes" BASEDIR="$SIM/ktg/loop" KTG_LOOP_SBATCH="$REAL_SBATCH" KTG_MIN_RUNTIME_SECONDS=1
# stand-in with the real script's semantics: bash -eu, pipefail, a pipeline stage like `katago | tee`
printf '#!/bin/bash -eu\nset -eu -o pipefail\necho "FAKE LOOP (set -eu -o pipefail in the body) ran: $*"\nsleep 30 | cat\nexit 0\n' > "$SIM/codes/loop/synchronous_loop_9x9.sh"; chmod +x "$SIM/codes/loop/synchronous_loop_9x9.sh"
reset_state() { rm -f "$BASEDIR/.failcount" "$BASEDIR/.chain_depth" "$BASEDIR/.breaker_tripped" "$BASEDIR/STOP"; : > "$VAL_SHIMLOG"; echo 700000 > "$VAL_JOBCTR"; }
state() { printf '.failcount=%s  .chain_depth=%s  breaker_tripped=%s  STOP=%s' "$(cat "$BASEDIR/.failcount" 2>/dev/null || echo '<absent>')" "$(cat "$BASEDIR/.chain_depth" 2>/dev/null || echo '<absent>')" "$([ -e "$BASEDIR/.breaker_tripped" ] && echo yes || echo no)" "$([ -e "$BASEDIR/STOP" ] && echo yes || echo no)"; }
filt() { sed -e "s#$SIM#\$SIM#g" -e "s#$AZ#\$AZ#g" -e 's/=== \[[^]]*\] /=== /' "$1" | grep -E '^=== (SIGTERM|loop exited|scheduler termination|clean exit|failure rc|fast failure|chain continues|circuit|cancelling|starting the loop)|^FAKE LOOP|Terminated'; }
shimlog() { echo "--- scheduler shim log"; if [ -s "$VAL_SHIMLOG" ]; then sed -e "s#$SIM#\$SIM#g" -e "s#$AZ#\$AZ#g" "$VAL_SHIMLOG"; else echo "(no sbatch, no scancel)"; fi; }
siglink() {
  local label="$1" jid="$2" mode="$3" pid lp rc i
  echo "----- link: $label   (mode=$mode, wrapper never signalled; KTG_MIN_RUNTIME_SECONDS=1)"
  SLURM_JOB_ID="$jid" setsid bash "$REAL_SBATCH" > "$SIM/out.txt" 2>&1 & pid=$!
  for i in $(seq 1 120); do grep -q 'starting the loop' "$SIM/out.txt" 2>/dev/null && break; sleep 0.5; done; sleep 1
  lp=$(pgrep -P "$pid" -f synchronous_loop_9x9.sh | head -1); echo "   loop bash pid found: ${lp:+yes}; its children: $(pgrep -P "$lp" | wc -l)"
  case "$mode" in
    child+loop) pkill -TERM -P "$lp"; kill -TERM "$lp" ;;
    loop-only)  kill -TERM "$lp" ;;
    child-only) pkill -TERM -P "$lp" ;;
  esac
  wait "$pid"; rc=$?; pkill -KILL -g "$pid" 2>/dev/null; pkill -KILL -P "$lp" 2>/dev/null
  filt "$SIM/out.txt"; echo "wrapper exit=$rc  $(state)"
}
echo "########## K4 redone (second pass): loop signalled, wrapper not (stand-in loop: set -eu -o pipefail IN THE BODY, 'sleep 30 | cat')  $(date -Is)"
echo "--- loop.sbatch sha256: $(sha256sum "$REAL_SBATCH" | cut -c1-16)...   real loop script header: $(head -2 "$LOOPDIR/synchronous_loop_9x9.sh" | tr '\n' ' ')"
reset_state; siglink K4a 9004 child+loop; shimlog
reset_state; siglink K4b 9014 loop-only; shimlog
reset_state; siglink K4c 9024 child-only; shimlog
