==== my_shims.sh ====
#!/bin/bash
# Validator's OWN fixture-driven scheduler shims (written independently of the worker's
# hp_common.sh). Fixture: $FIXNODES, one node per line:  <node> <partitions,comma> <State> <alloc_gpus>
# Mirrors what the REAL sinfo/scontrol on skipjack printed on 2026-09-04 (see validation record):
#   sinfo -p <list> -N -h -o %N   ignores empty tokens (',b200' 'b200,' 'b200,,l40s' all work),
#                                 prints nothing for an unknown partition, exit 0
#   scontrol show node <n>        State=<st> ... AllocTRES=cpu=..,mem=..,gres/gpu=<a>  (empty when idle)
# Usage: . my_shims.sh <bindir>
mk_shims() {
  local B="$1"
  cat > "$B/sinfo" <<'EOF'
#!/bin/bash
plist=""; while [ $# -gt 0 ]; do case "$1" in -p) plist="$2"; shift 2 ;; *) shift ;; esac; done
IFS=',' read -r -a want <<< "$plist"
while read -r n parts st a; do
  [ -n "${n:-}" ] || continue
  IFS=',' read -r -a np <<< "$parts"
  for q in "${want[@]}"; do
    [ -n "$q" ] || continue
    for pp in "${np[@]}"; do [ "$pp" = "$q" ] && echo "$n"; done
  done
done < "$FIXNODES"
exit 0
EOF
  cat > "$B/scontrol" <<'EOF'
#!/bin/bash
if [ "${1:-}" = show ] && [ "${2:-}" = node ]; then
  while read -r n parts st a; do
    if [ "$n" = "${3:-}" ]; then
      echo "NodeName=$n Arch=x86_64 CoresPerSocket=64 "
      echo "   CPUAlloc=$((a*16)) CPUEfctv=128 CPUTot=128 CPULoad=1.00"
      echo "   Gres=gpu:${parts%%,*}:8"
      echo "   State=$st ThreadsPerCore=1 TmpDisk=0 Weight=1 Owner=N/A MCS_label=N/A"
      echo "   Partitions=$parts "
      echo "   CfgTRES=cpu=128,mem=1547000M,billing=128,gres/gpu=8"
      if [ "$a" -gt 0 ]; then echo "   AllocTRES=cpu=$((a*16)),mem=$((a*64))G,gres/gpu=$a"; else echo "   AllocTRES="; fi
      exit 0
    fi
  done < "$FIXNODES"
fi
exit 0
EOF
  cat > "$B/sbatch" <<'EOF'
#!/bin/bash
n=$(cat "$VAL_JOBCTR" 2>/dev/null || echo 800000); n=$((n+1)); echo "$n" > "$VAL_JOBCTR"
echo "SBATCH[$n] argv: $*" >> "$VAL_SHIMLOG"
echo "SBATCH[$n] env : KTG_PARTITIONS='${KTG_PARTITIONS-<unset>}' SBATCH_PARTITION='${SBATCH_PARTITION-<unset>}' SBATCH_TIMELIMIT='${SBATCH_TIMELIMIT-<unset>}' SBATCH_EXPORT='${SBATCH_EXPORT-<unset>}' KTG_MAX_CHAIN='${KTG_MAX_CHAIN-<unset>}'" >> "$VAL_SHIMLOG"
echo "$n"
EOF
  chmod +x "$B/sinfo" "$B/scontrol" "$B/sbatch"
}
fixset() { : > "$FIXNODES"; }
fixadd() { printf '%s %s %s %s\n' "$1" "$2" "$3" "$4" >> "$FIXNODES"; }   # node partitions state alloc
fixshow() { echo "    fixture (node partitions State alloc_gpus; 8 GPUs per node):"; sed 's/^/      /' "$FIXNODES"; }
