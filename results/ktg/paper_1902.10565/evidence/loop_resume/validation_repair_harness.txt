link() {   # $1 label, $2 fake job id, $3 script (optional)
  local script="${3:-$REAL_SBATCH}" rc
  echo "----- link: $1   (KTG_BUDGET_ENV=${KTG_BUDGET_ENV:-<unset>})"

# Slurm-like termination: run the link in its own process group, wait until the loop
# has started, then signal.  $3 = target: pgrp | child-first | bash-only
siglink() {
  local label="$1" jid="$2" mode="$3" pid rc i
  echo "----- link: $label   (SIGTERM, mode=$mode)"
  SLURM_JOB_ID="$jid" setsid bash "$REAL_SBATCH" > "$SIM/out.txt" 2>&1 &
  pid=$!
  for i in $(seq 1 60); do grep -q 'starting the loop' "$SIM/out.txt" 2>/dev/null && break; sleep 0.5; done
  sleep 1
  case "$mode" in
    pgrp)        kill -TERM -- "-$pid" ;;
    child-first) pkill -TERM -P "$pid"; sleep 0.5; kill -TERM "$pid" ;;
    bash-only)   kill -TERM "$pid" ;;
  esac
  wait "$pid"; rc=$?
  pkill -TERM -f "$SIM/codes/loop/synchronous_loop_9x9.sh" 2>/dev/null
  filt "$SIM/out.txt"
  echo "wrapper exit=$rc  $(state)"
}
