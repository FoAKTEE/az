#!/bin/bash
# Sixth repair packet preamble (successor-id parsing past the site BILLING banner).
# It SOURCES the wave-3 validator's own w3v_common.sh VERBATIM (extracted from
# evidence/wave3_prelaunch_repairs/validation_harness.txt) so every shim, filter and
# accounting check below is the recorded one, and then replaces exactly TWO shims:
#
#   sbatch  -- modes selected by $R6_SBATCH_MODE, so the harness can reproduce what the
#              site's Slurm cli_filter actually does:
#                banner      "[BILLING] loaded!" + "[BILLING] Re-emitting cost at job
#                            start!" on STDERR, "<id>" on STDOUT, exit 0   <- the real site
#                banner-fed  the same banner, "<id>;skipjack" on STDOUT (the federated
#                            --parsable form), exit 0
#                clean       nothing on stderr, "<id>" on stdout, exit 0   <- w3v default
#                noise       banner on stderr, a site chatter line BEFORE the id on
#                            stdout, exit 0
#                fail        "sbatch: error: ..." on STDERR, nothing on stdout, exit 1
#                silent      nothing on either stream, exit 0  (the "no id, rc 0" anomaly)
#              Every mode logs "SBATCH[<n>] <args>" to $VAL_SHIMLOG exactly as the wave-3
#              shim does, so shimlog/cancel_check/state are unchanged.
#   squeue  -- $R6_SQUEUE_MODE=empty prints nothing (w3v behaviour: exit 0, no output);
#              =hit prints one PENDING row "<id>|afterany:<jobid>(unfulfilled)" in the
#              '%i|%E' shape the repaired lookup asks for, using the id the sbatch shim
#              would have handed back and the SLURM_JOB_ID of the running link.
#
# Nothing else is touched: same PATH, same fake loops, same reset_state/link/clink/
# siglink/state/filt/acct/cancel_check/shimlog helpers.
. "$(dirname "$0")/w3v_common.sh" "$1"

export R6_SBATCH_MODE="${R6_SBATCH_MODE:-banner}"
export R6_SQUEUE_MODE="${R6_SQUEUE_MODE:-empty}"

cat > "$SIM/bin/sbatch" <<'EOF'
#!/bin/bash
n=$(cat "$VAL_JOBCTR" 2>/dev/null || echo 800000); n=$((n+1)); echo "$n" > "$VAL_JOBCTR"
echo "SBATCH[$n] $*" >> "$VAL_SHIMLOG"
banner() { echo "[BILLING] loaded!" >&2; echo "[BILLING] Re-emitting cost at job start!" >&2; }
case "${R6_SBATCH_MODE:-banner}" in
  banner)     banner; echo "$n" ;;
  banner-fed) banner; echo "$n;skipjack" ;;
  clean)      echo "$n" ;;
  noise)      banner; echo "site-wrapper: submitting to partition"; echo "$n" ;;
  silent)     : ;;
  fail)       echo "sbatch: error: Batch job submission failed: Invalid account or account/partition combination specified" >&2; exit 1 ;;
  *)          banner; echo "$n" ;;
esac
exit 0
EOF
cat > "$SIM/bin/squeue" <<'EOF'
#!/bin/bash
# Only the resubmit lookup asks for the '%i|%E' shape; every other squeue caller in
# this harness is check.sh, which must keep seeing the wave-3 shim (exit 0, no output).
case "$*" in
  *'%i|%E'*) echo "SQUEUE $*" >> "$VAL_SHIMLOG"
             if [ "${R6_SQUEUE_MODE:-empty}" = hit ]; then
               n=$(cat "$VAL_JOBCTR" 2>/dev/null || echo 800000)
               echo "${n}|afterany:${SLURM_JOB_ID:-0}(unfulfilled)"
             fi ;;
esac
exit 0
EOF
chmod +x "$SIM/bin/sbatch" "$SIM/bin/squeue"
echo "r6 shims installed: sbatch(mode=\$R6_SBATCH_MODE) squeue(mode=\$R6_SQUEUE_MODE); all other shims are w3v_common.sh's"
echo
