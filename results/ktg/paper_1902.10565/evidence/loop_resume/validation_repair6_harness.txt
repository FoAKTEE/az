n=$(cat "$VAL_JOBCTR" 2>/dev/null || echo 800000); n=$((n+1)); echo "$n" > "$VAL_JOBCTR"

  echo "########## fixture $name : headers=$(grep -c '^== scratch_guard ' "$log") per-cycle=$(grep -c 'pre-gatekeeper\] ==' "$log") OK=$(grep -c 'scratch_guard: OK' "$log") WARNING=$(grep -c 'scratch_guard: WARNING' "$log") no-usable=$(grep -c 'scratch_guard: no usable' "$log")"
  for which in pre post; do s="$PRE"; [ $which = post ] && s="$POST"
    if [ "$name" = live-301099 ]; then base="$P1"; else base="$V/p10/base"; fi
    bash "$s" "$base" --log "$log" --no-append --no-du > "$V/p10/$name.$which.txt" 2>&1; rc=$?

    echo "  [$which] P10 verdict: $(grep -E '^(NOTICE|ESCALATE): *P10' "$V/p10/$name.$which.txt" || echo '(none)')"

ls -la "$R/evidence/production_chain/status_log.txt" | sed 's/^/    /'; echo "    (status_log.txt not appended to: --no-append)"
echo "done $(date -Is)"
