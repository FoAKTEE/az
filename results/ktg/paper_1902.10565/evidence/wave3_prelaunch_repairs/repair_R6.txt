#!/bin/bash
# Stand-in git tree + dummy stages, so the REAL synchronous_loop_9x9.sh can run a whole
# cycle under the REAL loop.sbatch on the login node with no engine, no trainer and no GPU.
# Copied from the fourth-repair validator harness ("val4a.sh source, verbatim", PART 2
# mkrepo / one_cycle form) and re-pointed so the wrapper, not the bare loop, drives it.
# Usage: source w3_real_common.sh <simname>   (after sourcing w3_common.sh <simname>)
mkrepo() { local d="$1"; rm -rf "$d"; mkdir -p "$d/python/selfplay" "$d/python/katago" "$d/python/muon" "$d/cpp/build"
  printf '# dummy\n' > "$d/python/a.py"; printf '# dummy\n' > "$d/python/b.py"; printf '# dummy\n' > "$d/python/selfplay/s.py"
  printf '#!/bin/bash\necho "DUMMY shuffle.sh $*"; exit 0\n' > "$d/python/selfplay/shuffle.sh"; chmod +x "$d/python/selfplay/shuffle.sh"
  printf '' > "$d/python/katago/__init__.py"; printf '' > "$d/python/muon/__init__.py"
  printf '#!/bin/bash\necho "DUMMY ENGINE $*"; exit 0\n' > "$d/cpp/build/katago"; chmod +x "$d/cpp/build/katago"
  (cd "$d" && git init -q && git config user.email w@x && git config user.name w && git add -A && git commit -q -m init) ; }

real_setup() {
  mkrepo "$SIM/repo"
  mkdir -p "$SIM/dummy"
  printf '#!/bin/bash\necho "DUMMY train_9x9.sh $*"; exit 0\n' > "$SIM/dummy/train_9x9.sh"
  printf '#!/bin/bash\necho "DUMMY export $*"; exit 0\n' > "$SIM/dummy/export_9x9.sh"
  printf '#!/bin/bash\necho "FASTGUARD ok"; exit 0\n' > "$SIM/dummy/fastguard.sh"
  chmod +x "$SIM"/dummy/*
  # the wrapper must launch the REAL loop script
  rm -f "$SIM/codes/loop/synchronous_loop_9x9.sh"
  ln -s "$REAL_LOOP" "$SIM/codes/loop/synchronous_loop_9x9.sh"
  printf 'export KATAGO_SRC=%s\nexport KATAGO_BIN=%s/cpp/build/katago\n' "$SIM/repo" "$SIM/repo" > "$SIM/ktg/env.sh"
  export TRAIN_WRAPPER="$SIM/dummy/train_9x9.sh" EXPORT_WRAPPER="$SIM/dummy/export_9x9.sh"
  export KTG_SCRATCH_GUARD="$SIM/dummy/fastguard.sh" KTG_PRUNE_RETENTION="$SIM/dummy/prune.py" KTG_ONE_CYCLE=1
  printf 'import sys\nprint("DUMMY prune_retention", sys.argv[1:])\n' > "$SIM/dummy/prune.py"
}
# one link of the REAL wrapper driving the REAL loop script for exactly one cycle
reallink() {
  local label="$1" jid="$2" rc
  echo "----- real link: $label"
  SLURM_JOB_ID="$jid" SLURM_CPUS_PER_TASK=32 bash "$REAL_SBATCH" > "$SIM/out.txt" 2>&1; rc=$?
  filt "$SIM/out.txt" | grep -E '=== |DUMMY |FASTGUARD|stage_monitor|cycle .* complete|pos_len|CHECK_POS_LEN|Gatekeeper|Selfplay|Shuffle|Train|Export|refus|FAIL' | sed 's/^/    /'
  echo "    wrapper exit=$rc  $(state)"
}
