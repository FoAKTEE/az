==== w3v_common.sh ====
#!/bin/bash
# Validator's own harness preamble for the wave-3 pre-launch repairs (R1/R5/R6) and the
# regression set: the wrapper codes/loop/loop.sbatch at HEAD under THIS harness's PATH
# shims (sbatch scancel sinfo sacct scontrol squeue nvidia-smi), stand-in loops, the real
# mission.json, the real check.sh, the real codes/data_budget, codes/cfg and codes/eval
# trees, a COPY of the real knobs_9x9.env. Login node, CPU only, no Slurm job, no GPU.
# Written independently of the worker's harness; the shim set and the accounting checks
# follow the fourth-repair validator record (validation_repair4_harness.txt).
# Usage: source w3v_common.sh <simname>
set -u
AZ=$HOME/az
RT=$RT
SIM="$RT/$1"
PAPER="$AZ/results/ktg/paper_1902.10565"
DB="$PAPER/codes/data_budget"
LOOPDIR="$PAPER/codes/loop"
CFGDIR="$PAPER/codes/cfg"
EVALDIR="$PAPER/codes/eval"
REAL_SBATCH="$LOOPDIR/loop.sbatch"
REAL_LOOP="$LOOPDIR/synchronous_loop_9x9.sh"
rm -rf "$SIM"; mkdir -p "$SIM/bin" "$SIM/ktg/loop" "$SIM/ktg/logs" "$SIM/codes/loop" "$SIM/src"
cat > "$SIM/bin/sbatch" <<'EOF'
#!/bin/bash
n=$(cat "$VAL_JOBCTR" 2>/dev/null || echo 800000); n=$((n+1)); echo "$n" > "$VAL_JOBCTR"
echo "SBATCH[$n] $*" >> "$VAL_SHIMLOG"; echo "$n"
EOF
cat > "$SIM/bin/scancel" <<'EOF'
#!/bin/bash
echo "SCANCEL $*" >> "$VAL_SHIMLOG"
EOF
for c in sinfo sacct scontrol squeue nvidia-smi; do printf '#!/bin/bash\nexit 0\n' > "$SIM/bin/$c"; done
chmod +x "$SIM"/bin/*
export VAL_SHIMLOG="$SIM/shim.log" VAL_JOBCTR="$SIM/jobctr"
export PATH="$SIM/bin:/usr/bin:$PATH"
ln -s "$DB" "$SIM/codes/data_budget"
ln -s "$CFGDIR" "$SIM/codes/cfg"
ln -s "$EVALDIR" "$SIM/codes/eval"
cp "$LOOPDIR/knobs_9x9.env" "$SIM/codes/loop/knobs_9x9.env"
printf 'export KATAGO_SRC=%s\nexport KATAGO_BIN=%s/katago\n' "$SIM/src" "$SIM/src" > "$SIM/ktg/env.sh"

# the loop's CHANGE 10 writer block, located by its own marker lines in the file under test
WB_START=$(grep -n '^    CYCLES_DONE=\$(cat "\$BASEDIR"/.cycles_completed' "$REAL_LOOP" | cut -d: -f1)
WB_END=$(grep -n '^    echo "cycle \$CYCLE_INDEX complete' "$REAL_LOOP" | cut -d: -f1)
awk -v a="$WB_START" -v b="$WB_END" 'NR>=a && NR<=b' "$REAL_LOOP" | sed -e 's/^    //' -e "s#\"\$BASEDIR\"/#$SIM/ktg/loop/#g" > "$SIM/writer_block.sh"

fake_loop() { printf '#!/bin/bash\necho "FAKE LOOP ran: $*"\n%s\n' "$1" > "$SIM/codes/loop/synchronous_loop_9x9.sh"; chmod +x "$SIM/codes/loop/synchronous_loop_9x9.sh"; }
cycle_loop() { { printf '#!/bin/bash\nset -eu -o pipefail\necho "FAKE LOOP ran: $*"\nCYCLE_INDEX=1\n'; cat "$SIM/writer_block.sh"; printf 'set +e\necho "FAKE LOOP recorded a cycle: $(cat %s/.cycles_completed)"\n%s\n' "$SIM/ktg/loop" "$1"; } > "$SIM/codes/loop/synchronous_loop_9x9.sh"; chmod +x "$SIM/codes/loop/synchronous_loop_9x9.sh"; }
eu_loop() { printf '#!/bin/bash -eu\nset -eu -o pipefail\necho "FAKE LOOP (set -eu -o pipefail in the body) ran: $*"\nsleep 30 | cat\nexit 0\n' > "$SIM/codes/loop/synchronous_loop_9x9.sh"; chmod +x "$SIM/codes/loop/synchronous_loop_9x9.sh"; }
# R6: a stand-in whose child renames itself to the engine's argv[0] so the sampler's classifier sees `katago selfplay`
mon_loop() { printf '#!/bin/bash\necho "FAKE LOOP (monitored) ran: $*"\n( exec -a "katago selfplay" sleep %s ) &\nwait\n%s\n' "$1" "${2:-exit 0}" > "$SIM/codes/loop/synchronous_loop_9x9.sh"; chmod +x "$SIM/codes/loop/synchronous_loop_9x9.sh"; }

export AZ_ROOT="$AZ" KTG_ROOT="$SIM/ktg" KTG_CODES="$SIM/codes" BASEDIR="$SIM/ktg/loop" KTG_LOOP_SBATCH="$REAL_SBATCH"
export SLURM_CPUS_PER_TASK=32

reset_state() { chmod -R u+rwx "$BASEDIR" 2>/dev/null; rm -rf "$BASEDIR/.failcount" "$BASEDIR/.chain_depth" "$BASEDIR/.breaker_tripped" "$BASEDIR/STOP" "$BASEDIR/.cycles_completed" "$BASEDIR/monitor"; : > "$VAL_SHIMLOG"; echo 800000 > "$VAL_JOBCTR"; }
showf() { local f="$1"; if [ -d "$f" ]; then echo '<dir>'; elif [ -e "$f" ]; then cat -A "$f" 2>/dev/null | tr '\n' '|' || echo '<unreadable>'; else echo '<absent>'; fi; }
state() { printf '.failcount=%s  .chain_depth=%s  .cycles_completed=%s  breaker_tripped=%s  STOP=%s' "$(showf "$BASEDIR/.failcount")" "$(showf "$BASEDIR/.chain_depth")" "$(showf "$BASEDIR/.cycles_completed")" "$([ -e "$BASEDIR/.breaker_tripped" ] && echo yes || echo no)" "$([ -e "$BASEDIR/STOP" ] && echo yes || echo no)"; }
# drops timestamps, tool chatter and the wave-3 INFORMATION lines so a regression diff sees only the accounting
filt() { sed -e "s#$SIM#\$SIM#g" -e "s#$AZ#\$AZ#g" -e 's/=== \[[^]]*\] /=== /' -e '/^== scratch_guard/,/^scratch_guard: /d' -e '/^== prune_retention/,/removed [0-9]* paths/{/removed [0-9]* paths/!d}' -e '/^== compute-budget self-check/,/^OK  *: request/d' -e '/^quotas.py /d' -e '/^   PROTECT /d' -e '/^-- /d' -e '/(nothing to remove)/d' -e '/^   *[a-z_]* *: /d' -e '/^=== loop knobs /d' -e '/^=== cpus granted /d' -e '/^=== cfg numGameThreads = /d' -e '/^stage_monitor: /d' -e '/^pos_len check: /d' "$1"; }
acct() { echo "accounting lines: $(grep -c -E 'failure rc=|clean exit|PRE-FLIGHT|scheduler termination at|deliberate chain stop|exit 0 before the loop' "$SIM/out.txt")"; }
fin_end() { if grep -q -E 'chain continues, |circuit breaker tripped after|cancelling queued successor|no queued successor to cancel|scheduler termination at walltime|clean exit -- failcount|exit 0 before the loop started|deliberate chain stop' "$SIM/out.txt"; then echo "finalize_end=yes"; else echo "finalize_end=NO"; fi; }
cancel_check() { local c t; c=$(grep -c SCANCEL "$VAL_SHIMLOG"); if grep -q -E 'circuit breaker tripped after|pre-flight failure -- the successor|storage guard stopped' "$SIM/out.txt"; then t=1; else t=0; fi
