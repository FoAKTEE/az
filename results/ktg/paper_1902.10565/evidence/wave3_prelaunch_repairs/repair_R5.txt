#!/usr/bin/env python3
"""Write a KataGo-shaped training .npz by hand (no numpy on the login node).

usage: mk_npz.py <out.npz> [--bad]   --bad gives binaryInputNCHWPacked the
       trailing shape (22, 21) of a pos_len 19 row set instead of (22, 11).
"""
import struct, sys, zipfile

GOOD = [("binaryInputNCHWPacked", "|u1", (1, 22, 11)),
        ("globalInputNC",        "<f4", (1, 19)),
        ("policyTargetsNCMove",  "<i2", (1, 2, 82)),
        ("globalTargetsNC",      "<f4", (1, 80)),
        ("scoreDistrN",          "|i1", (1, 282)),
        ("valueTargetsNCHW",     "|i1", (1, 5, 9, 9)),
        ("qValueTargetsNCMove",  "<i2", (1, 3, 82))]

def itemsize(d):
    return 1 if d[1] in "ub" and d[2:] == "1" else int(d[2:])

def npy(descr, shape):
    hdr = "{'descr': '%s', 'fortran_order': False, 'shape': (%s), }" % (
        descr, ", ".join(str(x) for x in shape) + ("," if len(shape) == 1 else ""))
    pad = (64 - (10 + len(hdr) + 1) % 64) % 64
    hdr = hdr + " " * pad + "\n"
    n = itemsize(descr)
    for d in shape:
        n *= d
    return b"\x93NUMPY" + struct.pack("<BBH", 1, 0, len(hdr)) + hdr.encode("latin1") + b"\x00" * n

def main(argv):
    out = argv[1]
    bad = "--bad" in argv
    with zipfile.ZipFile(out, "w") as zf:
        for name, descr, shape in GOOD:
            if bad and name == "binaryInputNCHWPacked":
                shape = (1, 22, 21)
            zf.writestr(name, npy(descr, shape))
    print("wrote %s (%s)" % (out, "pos_len 19 shape on binaryInputNCHWPacked" if bad else "pos_len 9"))

if __name__ == "__main__":
    sys.exit(main(sys.argv))
